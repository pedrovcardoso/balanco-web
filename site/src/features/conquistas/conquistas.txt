function shortName(val) {
  return String(val || 'Trilha')
    .replace(' PÃºblica', '')
    .replace(' e Controle Social', '')

}

function groupGameRows(rows) {
  const map = new Map();
  (Array.isArray(rows) ? rows : [])
    .filter(Boolean)
    .sort(
      (a, b) =>
        Number(a?.sectionOrder || 0) - Number(b?.sectionOrder || 0) ||
        Number(a?.phaseOrder || 0) - Number(b?.phaseOrder || 0)
    )
    .forEach((row) => {
      const key = row.sectionSlug || `secao-${row.gameSectionId || map.size + 1}`;
      if (!map.has(key)) {
        map.set(key, {
          slug: key,
          title: row.sectionTitle || key,
          order: Number(row.sectionOrder || map.size + 1),
          phases: [],
        });
      }
      map.get(key).phases.push(row);
    });
  return [...map.values()].sort((a, b) => a.order - b.order);
}

function getMedalCoinLevel(stars, max, unlocked) {
  if (!unlocked) return 'locked';
  if (stars >= Math.ceil(max * 0.75)) return 'gold';
  if (stars >= Math.ceil(max * 0.5)) return 'silver';
  return 'bronze';
}

function downloadBase64Pdf(base64, fileName) {
  const binary = window.atob(base64);
  const bytes = new Uint8Array(binary.length);
  for (let index = 0; index < binary.length; index += 1) {
    bytes[index] = binary.charCodeAt(index);
  }
  const url = URL.createObjectURL(new Blob([bytes], { type: 'application/pdf' }));
  const anchor = document.createElement('a');
  anchor.href = url;
  anchor.download = fileName;
  document.body.appendChild(anchor);
  anchor.click();
  anchor.remove();
  window.setTimeout(() => URL.revokeObjectURL(url), 1000);
}

export default function CertificateForm({ isLocked = true, participantName = 'Participante' }) {
  const [downloading, setDownloading] = useState(false);
  const [statusMessage, setStatusMessage] = useState('');

    if (isLocked || downloading) return;

        certificate.pdfBase64,

import api from '../../../lib/api.js';

export async function getConquistasData() {
  const [achievementsResponse, certificateResponse] = await Promise.all([
    api.get('/achievements'),
    api.get('/certificates/status'),
  ]);

  const payload = achievementsResponse.data || {};
  const achievements = Array.isArray(payload)
    ? payload
    : Array.isArray(payload.items)
      ? payload.items
      : [];

  return {
    achievements,
    summary: payload.summary || {
      total: achievements.length,
      unlocked: achievements.filter((item) => item?.unlocked).length,
    },
    certificate: certificateResponse.data || null,
  };
}

export async function issueCertificate(participantName, fiscalYear = new Date().getFullYear()) {
  const { data } = await api.post('/certificates/generate', { participantName, fiscalYear });
  return data;
}

export async function verifyCertificate(code) {
  const { data } = await api.get(`/certificates/verify/${encodeURIComponent(code)}`);
  return data;
}
