def serialize_date(value):
    if isinstance(value, (date, datetime)):
        return value.isoformat()
    return str(value) if value else None


def medal(*, key, title, medal_type, icon, ribbon, coin, unlocked, description, meta,
          current=None, target=None, awarded_at=None):
    return {
        'key': key,
        'title': title,
        'shortName': title,
        'type': medal_type,
        'icon': icon,
        'ribbon': ribbon,

async def get_achievements(db, uid):
    user_id = int(uid)

    sections = await db.fetch_all(
        "SELECT s.id AS sectionId, s.slug, s.title, s.order_index AS sectionOrder, "
        "COUNT(p.id) AS totalPhases, "
        "SUM(CASE WHEN g.completed_at IS NOT NULL THEN 1 ELSE 0 END) AS completedPhases, "
        "COALESCE(SUM(g.best_stars), 0) AS stars, "
        "MAX(g.completed_at) AS completedAt "
        "FROM tb_section s "
        "JOIN tb_phase p ON p.section_id=s.id AND p.is_active=1 "
        "LEFT JOIN tb_game_progress g ON g.phase_id=p.id AND g.user_id=%s "
        "WHERE s.is_active=1 "
        "GROUP BY s.id, s.slug, s.title, s.order_index "
        "ORDER BY s.order_index",
        (user_id,),
    )

    redemptions = await db.fetch_all(
        "SELECT item_key AS itemKey, redeemed_at AS redeemedAt "
        "FROM tb_trail_redemption WHERE user_id=%s",
        (user_id,),
    )
    redemption_map = {row['itemKey']: row for row in redemptions}

    certificate = await db.fetch_one(
        "SELECT validation_code AS verificationCode, issued_at AS issuedAt "
        "FROM tb_certificate WHERE user_id=%s ORDER BY issued_at DESC LIMIT 1",
        (user_id,),
    )

    medals = []
    total_stars = 0
    possible_stars = 0
    completed_sections = 0
    brilliant_count = 0

    def normalize(value):
        import unicodedata
        text = unicodedata.normalize('NFD', str(value or '').lower())
        return ''.join(char for char in text if unicodedata.category(char) != 'Mn')

    section_by_slug = {str(row.get('slug') or ''): row for row in sections}
    section_by_title = {normalize(row.get('title')): row for row in sections}
    used_section_ids = set()

    ordered_sections = []
    for legacy_slug, legacy_title, legacy_icon, legacy_ribbon in LEGACY_GAME_TRAILS:
        row = section_by_slug.get(legacy_slug) or section_by_title.get(normalize(legacy_title))
        ordered_sections.append((row, legacy_slug, legacy_title, legacy_icon, legacy_ribbon))
        if row:
            used_section_ids.add(row['sectionId'])

    for index, row in enumerate(sections):
        if row['sectionId'] in used_section_ids:
            continue
        icon, ribbon = SECTION_VISUALS[index % len(SECTION_VISUALS)]
        ordered_sections.append((row, row.get('slug') or str(row['sectionId']), row['title'], icon, ribbon))

    for row, fallback_slug, fallback_title, icon, ribbon in ordered_sections:
        total = int(row.get('totalPhases') or 0) if row else 0
        completed = int(row.get('completedPhases') or 0) if row else 0
        stars = int(row.get('stars') or 0) if row else 0
        maximum = total * 3
        unlocked = total > 0 and completed >= total
        brilliant = unlocked and maximum > 0 and stars >= maximum
        coin = 'gold' if brilliant else 'silver' if unlocked and stars >= maximum * .5 else 'bronze'
        title = row.get('title') if row else fallback_title
        slug = row.get('slug') if row else fallback_slug

        total_stars += stars
        possible_stars += maximum
        completed_sections += int(unlocked)
        brilliant_count += int(brilliant)

        medals.append(medal(
            key=f"game-section-{slug}",
            title=title,
            medal_type='Trilha de jogos',
            icon=icon,
            ribbon=ribbon,
            coin=coin,
            unlocked=unlocked,
            description=(
                f"Concedida por concluir todas as {total} fases da trilha {title}."
                if unlocked else
                f"Bloqueada. Conclua todas as fases da trilha {title}."
            ),
            meta=(
                f"{stars} de {maximum} estrelas" if unlocked else
                f"{completed}/{total} fases concluÃ­das" if total > 0 else
                "Trilha ainda nÃ£o cadastrada ou sem fases ativas"
            ),
            current=completed,

    for item_key, title, icon, ribbon, coin in TRAIL_DEFINITIONS:
        redemption = redemption_map.get(item_key)

        medals.append(medal(

async def create_or_get_user(db, cpf: str, name: str) -> dict:
    nome_clean = name.strip()[:150]
    async with db.transaction() as cur:
        await cur.execute(
            'SELECT id, cpf, nome FROM tb_user WHERE cpf=%s FOR UPDATE', (cpf,)
        )
        u = await cur.fetchone()
        if u:
            if u['nome'] != nome_clean:
                await cur.execute(
                    'UPDATE tb_user SET nome=%s WHERE id=%s',
                    (nome_clean, u['id']),
                )
                u['nome'] = nome_clean
        else:
            await cur.execute(
                'INSERT INTO tb_user (cpf, nome) VALUES (%s, %s)',
                (cpf, nome_clean),
            )
            u = {'id': cur.lastrowid, 'nome': nome_clean}

    user_id_str = str(u['id'])
    cpf_masked = f'***.{cpf[3:6]}.{cpf[6:9]}-**' if len(cpf) == 11 else None
    return {
        'userId': user_id_str,
        'publicUserId': user_id_str,
        'displayName': u['nome'],
        'cpfMasked': cpf_masked,
    }

async def eligibility(db, uid):
    user_id = int(uid)
    row = await db.fetch_one(
        "SELECT "
        "(SELECT COUNT(*) FROM tb_trail_redemption WHERE user_id=%s) AS redemptions, "
        "(SELECT COUNT(*) FROM tb_phase WHERE is_active=1) AS totalPhases, "
        "(SELECT COUNT(*) FROM tb_game_progress g "
        " JOIN tb_phase p ON p.id=g.phase_id "
        " WHERE g.user_id=%s AND p.is_active=1 AND g.best_score>=p.pass_score) AS completedPhases, "
        "(SELECT COUNT(*) FROM tb_certificate WHERE user_id=%s) AS certificateCount",
        (user_id, user_id, user_id),
    ) or {}

    redemptions = int(row.get('redemptions') or 0)
    total_phases = int(row.get('totalPhases') or 0)
    completed_phases = int(row.get('completedPhases') or 0)
    required_redemptions = 6

    return {
        'redemptions': redemptions,
        'requiredRedemptions': required_redemptions,
        'totalPhases': total_phases,
        'completedPhases': completed_phases,
        'remainingPhases': max(0, total_phases - completed_phases),
        'remainingRedemptions': max(0, required_redemptions - redemptions),
        'certificateCount': int(row.get('certificateCount') or 0),
        'eligible': (
            total_phases > 0
            and completed_phases >= total_phases
            and redemptions >= required_redemptions
        ),
    }

async def generate(db, uid, name, year):
    status = await eligibility(db, uid)
    if not status['eligible']:
        raise ValueError(
            f"Requisitos pendentes: {status['remainingPhases']} fase(s) e "
            f"{status['remainingRedemptions']} item(ns) da trilha."
        )

    existing = await db.fetch_one(
        "SELECT validation_code AS verificationCode, issued_at AS issuedAt "
        "FROM tb_certificate WHERE user_id=%s ORDER BY issued_at DESC LIMIT 1",
        (int(uid),),
    )
    if not existing:
        code = secrets.token_hex(8).upper()
        await db.execute(
            "INSERT INTO tb_certificate(user_id,validation_code) VALUES(%s,%s)",
            (int(uid), code),
        )
        existing = {
            'verificationCode': code,
            'issuedAt': datetime.now(timezone.utc).isoformat(),
        }

    return {
        **existing,
        'publicCertificateId': str(existing['verificationCode']),
        'pdfBase64': generate_certificate(
            name,
            year or datetime.now().year,
            existing['verificationCode'],
        ),
    }

import json

def parse(row):
    if not row:
        return row
    result = dict(row)
    for key in ("configJson", "optionsJson", "metadataJson"):
        if result.get(key) and isinstance(result[key], str):
            try:
                result[key] = json.loads(result[key])
            except (TypeError, ValueError, json.JSONDecodeError):
                pass
    return result

def import_from_json(json_file, mysql_conn):
    """Importa os dados de um arquivo JSON caso o SQL Server nÃ£o esteja acessÃ­vel."""
    print(f"[INFO] Importando dados do arquivo JSON: {json_file}...")
    if not os.path.exists(json_file):
        print(f"[ERRO] Arquivo {json_file} nÃ£o encontrado.")
        return

    with open(json_file, 'r', encoding='utf-8') as f:
        data = json.load(f)

    cursor_ms = mysql_conn.cursor()

    for table, records in data.items():
        if not records:
            continue
        print(f"[INFO] Importando {len(records)} registros para {table}...")
        cursor_ms.execute(f"TRUNCATE TABLE {table}")
        cols = list(records[0].keys())
        placeholders = ", ".join(["%s"] * len(cols))
        col_names = ", ".join(cols)
        sql = f"INSERT INTO {table} ({col_names}) VALUES ({placeholders})"
        vals = [[r.get(c) for c in cols] for r in records]
        cursor_ms.executemany(sql, vals)
        print(f"  â””â”€ Sucesso para {table}.")
